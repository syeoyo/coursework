\begin{frame}{분산에너지원 (Distributed Energy Resources, DER)}
    \textbf{에너지를 사용하는 공간ㆍ지역 또는 인근지역에서 공급하거나 생산하는 에너지로, 일정 규모 이하의 소규모 에너지 \footnotemark[1]}
    \begin{itemize}
        \item 태양광(PV) 및 풍력 터빈(WT) 발전 장치, 에너지 저장 시스템(ESS), 전기 자동차(EV) 충전기, 마이크로그리드 등을 포함
        \item 소규모 발전의 형태에 적합한 탄소중립적인 에너지를 중심으로 생산
    \end{itemize}

    \begin{block}{}
        $\Rightarrow$ 불안정한 대규모 중앙집중식 송전 및 발전에서 벗어나, 수요지 인근에서 환경친화적이고 독립적인 에너지의 생산과 소비 가능
    \end{block}

    \footnotetext[1]{대한민국 정부, 분산에너지 활성화 특별법 (2023), \url{https://www.law.go.kr/법령/분산에너지활성화특별법} \cite{Korea2023DER}}

\end{frame}

\begin{frame}{분산에너지원의 통합 (Aggregation)}
    \textbf{여러 개별 분산에너지원들을 하나의 포토폴리오(Aggregated Distributed Energy Resources, ADER)로 통합하는 과정으로, 가상 발전소나 에너지 관리 시스템 등을 통해 구현}
    \begin{block}{Aggregation이 필요한 이유}
        \textcircled{\small{1}} 재생에너지 발전량의 불확실성과 \textcircled{\small{2}} non-dispatchable한 출력 특성으로 인해,\\ 안정적인 day-ahead commitment를 하기 어렵고 그에 따라 페널티가 부과될 위험이 높음
    \end{block}

    \vspace{1em}

    \textbf{개별 DER들은 Aggregation을 통해}
    \begin{itemize}
        \item 서로의 발전잉여량과 계약미충족량을 상쇄시킴으로써 페널티 리스크 완화
        \item 하나의 포토폴리오로서 최적화된 day-ahead commitment와 real-time trading 수행
    \end{itemize}

    \begin{alertblock}{공정성에 대한 고려가 필요한 이유}
        포트폴리오 내에서 개별 DER의 에너지가 \textbf{불공정하게 활용}된다면, Aggregation에 참여할 유인이 감소하고 \textbf{참여자의 이탈}이 발생할 수 있음
        \begin{itemize}
            \item 페널티 발생 위험 증가
            \item 집합체 수준의 기대 수익 감소
            \item 개별적으로 정산받을 수 있는 양 감소
        \end{itemize}
        $\Rightarrow$ 포토폴리오 내의 개별 DER들 사이 공정성이 고려되어야 함
    \end{alertblock}

\end{frame}

\begin{frame}{방법론 소개}
    \begin{block}{Risk-Averse Two-stage Stochastic Optimization Model}
        \textbf{Objective Function} \\
        \begin{itemize}
            \item 수익 최대화 (하루전시장 수익 + 실시간시장 판매량 - 하루전시장 미충족 페널티)
            \item DER 간 기여도 편차를 리스크로 간주하여, 평균보다 적거나 많은 기여도 모두에 대해 페널티 부여
        \end{itemize}
        \textbf{First-Stage}: 불확실성 실현 \underline{이전}
        \begin{itemize}
            \item 하루전시장에 제출할 Commitment 양 $\left( \alpha, x_i \right)$
        \end{itemize}
        \textbf{Second-Stage}: 불확실성 실현 \underline{이후}
        \begin{itemize}
            \item 실시간시장에서의 판매량 / 계약미충족량 $\left( \beta(\xi), y_i(\xi)\right)$
            \item 에너지 저장 장치의 충$\cdot$방전량 $\left(z_i^C(\xi), z_i^D(\xi) \right)$
            \item Aggregation의 페널티 방지를 위한 기여량 $\left( d_i(\xi) \right)$
        \end{itemize}
    \end{block}
    \vspace{0.5em}
    \textbf{Sampling-based Approach}를 활용하여 실시간 시장 가격이나 발전량의 불확실성 고려 \\
    $\rightarrow$ 각 시나리오들은 historical data를 바탕으로 구성되며, second-stage decision
    variable들과 stochastic parameter들은 scenario-dependent하게 $\xi_s$ 로 재정의됨

\end{frame}

\begin{frame}{공정성 기반 위험 측정 (Fairness-Aware Risk Factor)}
    각 DER \(i\)의 $T$시간 동안의 시나리오$(\xi_s)$별 총 기여도(금액):
    \[
        \Lambda_{i}(\xi_s) = \sum_{t \in T} d_{it}(\xi_s) \cdot (P^{PN}_t - P^{RT}_{t}(\xi_s))
    \]
    \vspace{-1em}
    \begin{itemize}
        \item 각 DER이 실시간 시장에서 판매하여 얻을 수 있던 수익 중 일부($d \cdot P^{RT}$)를, 포트폴리오 전체의 페널티 회피($d \cdot P^{PN}$)를 위해 양보했다고 해석
        \item 개인의 기회비용으로서 정의되는 marginal contribution
    \end{itemize}
    \vspace{-0.5em}
    \begin{alertblock}{Central Deviation 활용}
        포트폴리오의 공정성을 확보하기 위해, 각 DER의 기여도 금액이 평균에서 벗어나는 경우에 대해 리스크로 간주하는 \textbf{양방향 절대편차 기반 측도} 도입
        \[
            \phi_{\text{CDEV}}(\Lambda_i) := \mathbb{E}_s \big[ \left| \Lambda_i(\xi_s) - \mathbb{E}_i[\Lambda_i(\xi_s)] \right| \big]
        \]
        \vspace{-1.5em}
        \begin{itemize}
            \item 과도하게 기여하거나 충분히 기여하지 않은 DER 모두를 공정성 위반으로 간주
            \item 공정성은 전체 시나리오에서의 기여 편차를 지속적으로 관리해야 실질적으로 보장되므로, tail risk만 반영하는 quantile risk measure보다 deviation-based 측도가 더 적합하다고 판단
        \end{itemize}
    \end{alertblock}

\end{frame}

\begin{frame}{결정변수 및 파라미터 정의}
    \small
    \begin{minipage}[t]{0.37\textwidth}
        \textbf{인덱스}
        \begin{itemize}

            \item $t \in T$: 시간 인덱스 (1시간 단위)

            \item $P_t^{RT}(\xi)$: 시점 $t$에서 실시간시장 전력 가격
            \item $P_t^{PN}$: 시점 $t$에서 계약 불이행 시 부과되는 페널티 가격
            \item $R_{it}(\xi)$: 시점 $t$에서 DER $i$의 발전량
            \item $K_i$: DER $i$의 배터리 용량
            \item $M_1, M_2$: Big-M 계수
            \item $\lambda$: 공정성 리스크의 가중치
        \end{itemize}
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.58\textwidth}
        \textbf{결정변수}
        \begin{itemize}
            \item $x_{it}$: 시점 $t$에서 DER $i$의 하루전시장 전력 거래량
            \item $y_{it}^{+}(\xi)$: 시점 $t$에서 DER $i$의 실시간시장 추가 판매량
            \item $y_{it}^{-}(\xi)$: 시점 $t$에서 DER $i$의 commitment 미충족량
            \item $z_{it}(\xi)$: 시점 $t$에서 DER $i$의 배터리 잔량
            \item $z^C_{it}(\xi), z^D_{it}(\xi)$: 시점 $t$에서 DER $i$이 충$\cdot$방전한 에너지 양
            \item $d_{ijt}(\xi)$: 시점 $t$에서 DER $i$가 $j$의 페널티 상쇄를 위해 활용된 에너지 양
            \item $e^+_{it}(\xi)$: 내부조정 후 실시간시장에 판매 가능한 잉여 에너지
            \item $e^-_{it}(\xi)$: 내부조정 후 남은 페널티 양
            \item $\Lambda_i(\xi)$: DER $i$의 시나리오별 총 기여 금액
            \item $\Delta_i(\xi)$: DER $i$의 평균 기여 대비 절대편차
            \item $\phi^1_{it}(\xi), \phi^2_{it}(\xi), \phi^3_{it}(\xi), \phi^4_{t}(\xi)$: Big-M 기반 이진 변수들
        \end{itemize}
    \end{minipage}
\end{frame}

\begin{frame}{수리 모델: Risk-Neutral 모델 (1)}
    \vspace{-1em}
    \footnotesize
    \begin{block}{}
        \vspace{-1em}
        \begin{subequations}
            \begin{flalign}
                \text{max} \;                     & \sum_{t \in T} \left(
                P_t^{DA} \sum_{i \in I} x_{it}
                + \mathbb{E}\left[
                P_t^{RT}(\xi_s) \sum_{i \in I} e^+_{it}(\xi_s)
                - P_t^{PN} \sum_{i \in I} e^-_{it}(\xi_s)
                \right]
                \right) \tag{1a}                                                                                                                                                        \\[0.5em]
                                                  & R_{it}(\xi_s) - x_{it} = y_{it}^{+}(\xi_s) - y_{it}^{-}(\xi_s) + z^C_{it}(\xi_s) - z^D_{it}(\xi_s) \quad & \forall i, t, s \tag{1b} \\
                                                  & R_{it}(\xi_s) \geq y^+_{it}(\xi_s) \quad                                                                 & \forall i, t, s \tag{1c} \\
                                                  & z_{i,t+1}(\xi_s) = z_{it}(\xi_s) + z^C_{it}(\xi_s) - z^D_{it}(\xi_s) \quad                               & \forall i, t, s \tag{1d} \\
                                                  & z^D_{it}(\xi_s) \le z_{it}(\xi_s), \quad
                z^C_{it}(\xi_s) \le K_i - z_{it}(\xi_s), \quad
                0 \le z_{it}(\xi_s) \le K_i \quad & \forall i, t, s \tag{1e}                                                                                                            \\
                                                  & e^+_{it}(\xi_s) = y^+_{it}(\xi_s) - \sum_{j \in I} d_{ijt}(\xi_s), \quad
                e^-_{it}(\xi_s) = y^-_{it}(\xi_s) - \sum_{j \in I} d_{jit}(\xi_s)
                                                  & \forall i, t, s \tag{1f}                                                                                                            \\
                                                  & d_{iit}(\xi_s) = 0
                                                  & \forall i, t, s \tag{1g}
            \end{flalign}
        \end{subequations}
    \end{block}

    \small
    \begin{itemize}
        \item[(1a)] 포토폴리오의 하루전시장 + 실시간시장 수익의 기대값을 최대화
        \item[(1b)] 발전량은 day-ahead 이행량, 실시간 조정량, 충방전량의 합으로 설명됨
        \item[(1c)] 실시간 판매량은 실제 발전량 이하로 제한
        \item[(1d)] 배터리 잔량은 시간에 따라 이월되며 충/방전량에 의해 갱신됨
        \item[(1e)] 배터리 용량 내에서의 충전 가능량, 방전 가능량, 저장한계 제약 반영
        \item[(1f)] 내부 조정의 결과로 에너지 잉여량 및 페널티 대상량이 다시 결정됨
        \item[(1g)] 자기 자신과의 내부 거래는 불가능하도록 설정
    \end{itemize}
\end{frame}

\begin{frame}{수리 모델: CDEV 기반 Risk-Averse 모델}
    \vspace{-1em}
    \begin{block}{}
        \vspace{-1em}
        \begin{subequations}
            \scriptsize
            \begin{flalign}
                \text{max} \quad
                 & \sum_{t \in T} \left(
                P_t^{DA} \sum_{i \in I} x_{it}
                + \mathbb{E}\left[
                P_t^{RT}(\xi_s) \sum_{i \in I} e^+_{it}(\xi_s)
                - P_t^{PN} \sum_{i \in I} e^-_{it}(\xi_s)
                \right]
                \right)
                - \lambda \cdot \mathbb{E} \left[ \sum_{i \in I} \Delta_i(\xi_s) \right]                                             \tag{2a}                             \\[0.5em]
                \text{s.t.} \quad
                 & \text{Constraints (1b)--(1m) are preserved} \nonumber                                                                                                  \\[0.5em]
                 & \Lambda_i(\xi_s) = \sum_{t \in T} \sum_{j \ne i} d_{ijt}(\xi_s) \cdot (P_t^{PN} - P_t^{RT}(\xi_s))                             & \forall i, s \tag{2b} \\
                 & \bar{\Lambda}(\xi_s) = \frac{1}{I} \sum_{i \in I} \Lambda_i(\xi_s)                                                             & \forall s   \tag{2c}  \\
                 & \Delta_i(\xi_s) \ge \Lambda_i(\xi_s) - \bar{\Lambda}(\xi_s), \quad \Delta_i(\xi_s) \ge \bar{\Lambda}(\xi_s) - \Lambda_i(\xi_s) & \forall i, s \tag{2d} \\
                 & \lambda \ge 0                      \tag{2e}                                                                                    &
            \end{flalign}
        \end{subequations}
    \end{block}
    \small
    \begin{itemize}
        \item[(2a)] 포토폴리오의 기대 수익을 최대화하되, 기여도 편차에 따른 공정성 리스크를 페널티로 부여
        \item[(2b)] DER $i$의 총 기여 금액은 포토폴리오의 페널티 상쇄를 위해 제공한 에너지를 기회비용으로 책정
        \item[(2c)] 해당 시나리오에서의 DER 전체 평균 기여 금액 계산
        \item[(2d)] DER $i$의 기여도가 평균보다 크거나 작을 때의 절대편차를 정의
        \item[(2e)] $\lambda$는 공정성 리스크의 민감도를 조절하는 계수로, 음수가 될 수 없음
    \end{itemize}
\end{frame}

\begin{frame}{질문 답변}
    \small
    \begin{itemize}
        \item[Q1.] 실험을 통해 나온 모델이 적용 가능한 모델인가요?
        \begin{itemize}
            \item[A1.] 기술적 제약(배터리 효율성, 충방전 손실, 수명 등)과 네트워크 제약(전력망 안정성, 전압 제한, 전송 용량)이 고려되어야하며, aggregation이 개별모델보다 높은 수익성을 항시 보장하지 않는다는 점에서 보완해야 합니다.
        \end{itemize}
        \item[Q2.] 과도하게 기여한 것보다 충분히 기여하지 않은 DER에 더 큰 패널티를 부과해야하지 않을까요?
        \begin{itemize}
            \item[A2.] 실제로는 부족 기여가 더 큰 문제를 야기하기에, 비대칭적 페널티 체계(부족 기여에 더 큰 가중치)를 도입하는 것이 더 현실적입니다. 하지만, 부족 기여에 대한 페널티를 가중하여 부여한다하더라도, 결국 최적화 과정에서 시스템은 리스크를 최소화하기 위해 모든 DER가 평균에 맞춰 기여하도록 유도하게 됩니다.
        \end{itemize}
        \item[Q3.] 시계열(발전량 데이터, 실시간 시장가)에 노이즈를 추가하는 것이 적절한 시나리오 생성 방법인가요?
        \begin{itemize}
            \item[A3.] 둘 다 변동가능성이 크다는 점에서 노이즈를 추가하여 시나리오를 생성하는 것이 적절하다고 생각하지만, 발전량과 실시간가격 간의 상관관계를 노이즈 생성 과정에 반영해야 할지 추가 검토가 필요합니다.
        \end{itemize}
        \item[Q4.] 재생에너지의 경우 가까운 지역에서는 움직임이 비슷해서 hedging의 효과가 크지 않을 것 같은데, 어느 정도의 효과를 기대해 볼 수 있나요?
        \begin{itemize}
            \item[A4.] 실제로 가까운 지역에서는 발전량 패턴이 유사하여 hedging의 효과가 크지 않으며, 발전량 패턴이 다른 경우의 aggregation 효과는 더 커집니다.
        \end{itemize}
    \end{itemize}
\end{frame}
